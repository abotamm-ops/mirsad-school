      }
      if(path.startsWith('/_#/')){raw=origin+path.slice(3);continue}
      if(!/^\/dl\/launcher\/launcher\.html\?/.test(path))return '';
      const targets=[];

      if(targets.length!==1)return '';
      let target=targets[0];
      try{for(let n=0;n<2&&target.startsWith('%');n++)target=decodeURIComponent(target)}catch{return ''}
      if(target.startsWith('/')&&!target.startsWith('//'))raw=origin+target;

      else return '';
    }

  }

/* Shared deterministic safety rules; no attendance mutation or timer ownership. */
var MirsadVideoCore=(()=>{
 const SAFETY_MS=30000,MIN_CLIP_MS=5000,MAX_CLIP_MS=600000;
 const normalize=s=>String(s||'').normalize('NFKC').replace(/\s+/g,' ').trim();
 function gap(run,now,minBudget=MIN_CLIP_MS,tail=null){
  if(!run?.running||run.paused||run.testMode||run.backtestMode||run.immediateMode||run.calendarTrial||run.current||run.meetingScan||!Number.isInteger(run.cycle)||run.cycle<1||run.cycle>3)return null;
  if(run.waitingForCycle2&&run.cycle===3)return null;const final= !run.waitingForCycle2;if(final&&(!tail||tail.startedAt!==run.startedAt||(tail.controlEpoch||0)!==(run.controlEpoch||0)||tail.period!==run.period||tail.cycle!==run.cycle||!run.periodItems?.length||run.index<run.periodItems.length))return null;const boundary=final?tail.endAt:run.nextCycleAt;if(!Number.isFinite(boundary))return null;const deadline=boundary-SAFETY_MS,budget=deadline-now;
  if(budget<minBudget||budget>12*60*60*1000)return null;
  return {period:Number(run.period),afterCycle:run.cycle,nextCycleAt:boundary,boundary:final?'lessonEnd':'nextCycle',deadline,budget,key:[run.startedAt,run.controlEpoch||0,run.period,run.cycle,boundary].join('|')};
 }
 function settings(input,roundCount){
  if(!Number.isInteger(input.minutes)||input.minutes<1||input.minutes>10)throw Error('اختر مدة من دقيقة إلى 10 دقائق');
  if(!Array.isArray(input.cycles)||input.cycles.some(c=>!Number.isInteger(c)||c<1||c>roundCount))throw Error('اختر دورة من الدورات المتاحة للحصة');
  return {...input,cycles:[...new Set(input.cycles)].sort(),enabled:input.cycles.length>0};
 }
 function budget(run,job,now,tail=null){const g=gap(run,now,0,tail);if(!g||g.key!==job.gapKey||g.period!==job.period||g.afterCycle!==job.afterCycle)return 0;return Math.max(0,Math.min(job.minutes*60000,g.deadline-now,Number(job.deadline)-now,MAX_CLIP_MS))}
 function shouldStop(run,job,now){return budget(run,job,now)<1}
 function safeName(value){return normalize(value).replace(/[\\/:*?"<>|\u0000-\u001f]/g,'_').replace(/[ .]+$/g,'').slice(0,90)||'معلم'}
 function fileName(job){return safeName(job.teacherName)+'_ح'+job.period+'_بعد'+job.afterCycle+'_'+job.date+'_'+job.id+'.webm'}
 return {SAFETY_MS,MIN_CLIP_MS,MAX_CLIP_MS,gap,settings,budget,shouldStop,safeName,fileName};
})();

function schoolChangeRequest_(p, props) {
  var key=props.getProperty('MONITOR_API_KEY');
  if(!key||key.length<32||!monitorEqual_(p.key,key))throw new Error('رمز ربط المدارس غير صالح');
  var active=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACTIVE')||'null');
  var pending=JSON.parse(props.getProperty('MIRSAD_SCHOOL_CHANGE')||'null');
  var last=JSON.parse(props.getProperty('MIRSAD_SCHOOL_LAST_CHANGE')||'null');
  if(p.action==='schoolChangeStatus')return {schoolId:active?.schoolId||null,generation:active?.generation||null,pending:pending?.id||null};
  if(p.action==='schoolChangePrepare'){
    var id=String(p.nextSchoolId||'').toLowerCase();
    if(!/^[a-f0-9]{32}$/.test(id)||!['disk','delete'].includes(p.disposition)||p.confirmed!==true)throw new Error('أكّد اختيار المدرسة ومصير السجلات');
    if(pending){
      if(pending.schoolId!==id||pending.disposition!==p.disposition)throw new Error('هناك عملية تبديل جارية');
      return {id:pending.id,hasPreviousSchool:!!pending.previousRoot,committed:false};
    }
    var lease=JSON.parse(props.getProperty('MIRSAD_SCHOOL_LEASE')||'null');
    if(lease&&lease.until>Date.now())throw new Error('أوقف الرصد على الجهاز المشغّل قبل تبديل المدرسة');
    if(active?.schoolId===id)return {id:last?.id||active.generation,committed:true};
    var root=DriveApp.getFolderById(MONITOR_ROOT);
    if(root.isTrashed()||root.getSharingAccess()!==DriveApp.Access.PRIVATE)throw new Error('مجلد المدرسة غير متاح أو غير خاص');
    pending={id:Utilities.getUuid(),schoolId:id,disposition:p.disposition,previousRoot:MONITOR_ROOT};
    props.setProperty('MIRSAD_SCHOOL_CHANGE',JSON.stringify(pending));
    return {id:pending.id,hasPreviousSchool:true,committed:false};
  }
  if(p.action==='schoolChangeCommit'&&last?.id===p.transitionId&&!pending)return {committed:true,generation:last.id};
  if(!pending||pending.id!==p.transitionId)throw new Error('طلب تبديل المدرسة غير صالح');
  if(p.action==='schoolChangeCancel'){
    if(pending.nextRoot)throw new Error('بدأ التبديل؛ أعد محاولة إكماله');
    props.deleteProperty('MIRSAD_SCHOOL_CHANGE');
    return {cancelled:true};
  }
  if(p.action==='schoolChangeExport'||p.action==='schoolChangeExportChunk'){
    if(pending.disposition!=='disk'||pending.nextRoot)throw new Error('التصدير غير متاح لهذه العملية');
    if(p.action==='schoolChangeExportChunk')return schoolExportChunk_(p,pending);
    var files=[],stack=[{folder:DriveApp.getFolderById(pending.previousRoot),path:[]}];
    while(stack.length){
      var entry=stack.pop(),children=entry.folder.getFolders(),entries=entry.folder.getFiles();
      while(children.hasNext()){
        var child=children.next();
        stack.push({folder:child,path:entry.path.concat({id:child.getId(),name:child.getName()})});
      }
      while(entries.hasNext()){
        var file=entries.next();
        files.push({id:file.getId(),name:file.getName(),size:file.getSize(),path:entry.path});
      }
    }
    if(!pending.exportId){pending.exportId=Utilities.getUuid();props.setProperty('MIRSAD_SCHOOL_CHANGE',JSON.stringify(pending))}
    return {exportId:pending.exportId,schoolId:active?.schoolId||null,files:files};
  }
  if(p.action!=='schoolChangeCommit')throw new Error('طلب غير مدعوم');
  if(pending.disposition==='disk'&&(!pending.exportId||p.receipt?.exportId!==pending.exportId||p.receipt?.transitionId!==pending.id||p.receipt?.saved!==true))
    throw new Error('لم يتأكد تصدير سجلات المدرسة إلى القرص');
  if(!pending.nextRoot){
    var next=DriveApp.createFolder('توثيق غياب مرصاد المدارس');
    next.setSharing(DriveApp.Access.PRIVATE,DriveApp.Permission.NONE);
    pending.nextRoot=next.getId();
    props.setProperty('MIRSAD_SCHOOL_CHANGE',JSON.stringify(pending));
  }
  var previous=DriveApp.getFolderById(pending.previousRoot);
  if(!previous.isTrashed())previous.setTrashed(true);
  var selection={schoolId:pending.schoolId,generation:pending.id};
  props.setProperties({
    MIRSAD_SCHOOL_DRIVE_ROOT:pending.nextRoot,
    MIRSAD_SCHOOL_ACTIVE:JSON.stringify(selection),
    MIRSAD_SCHOOL_LAST_CHANGE:JSON.stringify({id:pending.id})
  });
  props.deleteProperty('MIRSAD_SCHOOL_CHANGE');
  return {committed:true,generation:pending.id};
}

function schoolExportChunk_(p,pending){
  if(!pending.exportId||p.exportId!==pending.exportId||!/^[-_a-zA-Z0-9]+$/.test(p.fileId||''))throw new Error('ملف الأرشيف غير صالح');
  var file=DriveApp.getFileById(p.fileId), parents=file.getParents(), seen={};
  var within=false;
  while(parents.hasNext()){
    var parent=parents.next();
    while(parent&&!seen[parent.getId()]){
      var id=parent.getId();
      if(id===pending.previousRoot){within=true;break}
      seen[id]=true;
      var ancestors=parent.getParents();
      parent=ancestors.hasNext()?ancestors.next():null;
    }
    if(within)break;
  }
  if(!within||file.isTrashed())throw new Error('الملف خارج مجلد المدرسة');
  var size=file.getSize(),start=p.start;
  if(!Number.isSafeInteger(start)||start<0||start>=size)throw new Error('نطاق ملف الأرشيف غير صالح');
  var end=Math.min(start+2*1024*1024,size)-1;
  var reply=UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/'+encodeURIComponent(file.getId())+'?alt=media',{
    headers:{Authorization:'Bearer '+ScriptApp.getOAuthToken(),Range:'bytes='+start+'-'+end},muteHttpExceptions:true
  });
  var status=reply.getResponseCode();
  if(status!==206&&!(status===200&&start===0&&end===size-1))throw new Error('تعذر قراءة جزء من ملف الأرشيف');
  if(status===206){
    var headers=reply.getAllHeaders();
    var rangeKey=Object.keys(headers).find(function(name){return name.toLowerCase()==='content-range'});
    if(!rangeKey||String(headers[rangeKey])!=='bytes '+start+'-'+end+'/'+size)throw new Error('نطاق الجزء المستلم غير صحيح');
  }
  var bytes=reply.getContent();
  if(bytes.length!==end-start+1)throw new Error('حجم الجزء المستلم غير صحيح');
  return {content:Utilities.base64Encode(bytes),next:end+1};
}

function schoolSiteCallback(action,input){
  var lock=LockService.getScriptLock();
  lock.waitLock(30000);
  try{
    var props=PropertiesService.getScriptProperties(), now=Date.now();
    input=input||{};
    if(action==='login'){
      var state=JSON.parse(props.getProperty('MIRSAD_SITE_LOGIN_LIMIT')||'null');
      if(!state||state.until<=now)state={until:now+60000,count:0};
      if(state.count>=6)return {ok:false,error:'انتظر دقيقة ثم أعد المحاولة'};
      var pin=String(input.pin||'');
      var expected=props.getProperty('MONITOR_SITE_PASSWORD');
      var school=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACTIVE')||'null');
      var account=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACCOUNT')||'null');
      var signInEmail=String(input.email||'').trim().toLowerCase();
      if((account&&account.scope!==(school?.schoolId||'initial'))||
         (account&&account.email!==signInEmail)||!/^\d{4}$/.test(pin)||!expected||!monitorEqual_(pin,expected)){
        state.count++;
        props.setProperty('MIRSAD_SITE_LOGIN_LIMIT',JSON.stringify(state));
        return {ok:false,error:'الرقم السري غير صحيح أو لم يُضبط بعد'};
      }
      props.deleteProperty('MIRSAD_SITE_LOGIN_LIMIT');
      return {ok:true,...schoolSiteMint_(props)};
    }
    if(!/^[a-f0-9-]{72}$/i.test(input.session||''))return {ok:false,login:true};
    var key='MIRSAD_SITE_SESSION_'+schoolSiteHash_(input.session);
    if(action==='logout'){props.deleteProperty(key);return {ok:true}}
    if(!['day','announcement'].includes(action))return {ok:false,error:'طلب عرض غير مدعوم'};
    var session=JSON.parse(props.getProperty(key)||'null'),active=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACTIVE')||'null');
    if(!session||session.until<=now||session.generation!==(active?.generation||null))return {ok:false,login:true};
    if(props.getProperty('MIRSAD_SCHOOL_CHANGE'))return {ok:false,error:'تبديل المدرسة جارٍ'};
    if(action==='announcement'){
      var announcement=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ANNOUNCEMENT')||'null');
      return {ok:true,announcement:announcement?.scope===(active?.schoolId||'initial')?announcement:null};

      return {name:record.teacherName,period:record.period,cycle:record.cycle,status:record.status};
    })};
  }finally{lock.releaseLock()}
}

  if(active&&p.schoolGeneration!==active.generation)throw new Error('تغيرت المدرسة. حدّث الربط أولًا');
  if(props.getProperty('MIRSAD_SCHOOL_CHANGE'))throw new Error('أكمل تبديل المدرسة أولًا');
  var account=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACCOUNT')||'null');
  var scope=active?.schoolId||'initial';
  if(p.action==='schoolAccountInfo')return {email:account?.scope===scope?account.email:null};
  if(p.action==='schoolSettingsRead'||p.action==='schoolAnnouncementSave'){
    var announcement=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ANNOUNCEMENT')||'null');
    if(p.action==='schoolAnnouncementSave'){
      var title=String(p.title||'').trim(),body=String(p.body||'').trim();
      if(title.length>120||body.length>3000||(!body&&title))throw Error('راجع عنوان الإعلان ونصه');
      announcement=body?{scope:scope,title:title,body:body}:null;
      if(announcement)props.setProperty('MIRSAD_SCHOOL_ANNOUNCEMENT',JSON.stringify(announcement));
      else props.deleteProperty('MIRSAD_SCHOOL_ANNOUNCEMENT');
      return {announcement:announcement};
    }
    var rootId=props.getProperty('MIRSAD_SCHOOL_DRIVE_ROOT'),date=monitorToday_();
    var folder=rootId?monitorScheduleFolder_(DriveApp.getFolderById(rootId),date,false):null;
    var data=folder?monitorRead_(monitorFile_(folder,'attendance.json'),null):null;
    return {date:date,announcement:announcement?.scope===scope?announcement:null,lessons:(data?.snapshot?.lessons||[]).map(function(lesson){
      return {name:lesson.teacherName,period:lesson.period,calendarUrl:lesson.calendarUrl||null};
    })};
  }
  var email=String(p.email||'').trim().toLowerCase(),now=Date.now();
  if(email.length>254||!/^\S+@[^\s@]+\.[^\s@]+$/.test(email)||email.split('@').length!==2)throw new Error('أدخل بريد المدرسة الصحيح');
  if(account?.scope===scope&&account.email!==email)throw new Error('هذا الموقع مفعّل ببريد مدرسة آخر');
  var saved=JSON.parse(props.getProperty('MIRSAD_SCHOOL_ACTIVATION')||'null');
  if(p.action==='schoolAccountCode'){
    if(saved&&now-saved.at<60000)throw new Error('انتظر دقيقة قبل طلب رسالة ثانية');
    if(MailApp.getRemainingDailyQuota()<1)throw new Error('بلغت خدمة البريد حد الإرسال اليوم');
    var hex=Utilities.getUuid().replace(/-/g,'').slice(0,8);
    var code=String(100000+(parseInt(hex,16)%900000));
    props.setProperty('MIRSAD_SCHOOL_ACTIVATION',JSON.stringify({email:email,scope:scope,hash:schoolSiteHash_(code+'|'+writer),at:now,tries:0}));
    try{MailApp.sendEmail(email,'تفعيل مرصاد المدارس','رمز تفعيل مرصاد: '+code+'\nصالح لمدة عشر دقائق. أدخله في قسم الحساب بالإضافه.')}
    catch(error){props.deleteProperty('MIRSAD_SCHOOL_ACTIVATION');throw error}
    return {sent:true};
  }
  var code=String(p.code||''),pin=String(p.pin||'');
  if(!/^\d{4}$/.test(pin))throw new Error('اختر رقمًا سريًا من أربعة أرقام');
  if(!saved||saved.email!==email||saved.scope!==scope||now-saved.at>=600000||saved.tries>=5)throw new Error('اطلب رمز تفعيل جديدًا');
  if(!/^\d{6}$/.test(code)||!monitorEqual_(saved.hash,schoolSiteHash_(code+'|'+writer))){
    saved.tries++;
    props.setProperty('MIRSAD_SCHOOL_ACTIVATION',JSON.stringify(saved));
    throw new Error('رمز التفعيل غير صحيح');
  }
  props.setProperties({MIRSAD_SCHOOL_ACCOUNT:JSON.stringify({email:email,scope:scope}),MONITOR_SITE_PASSWORD:pin});
  props.deleteProperty('MIRSAD_SCHOOL_ACTIVATION');
  Object.keys(props.getProperties()).filter(function(key){return key.startsWith('MIRSAD_SITE_SESSION_')}).forEach(function(key){props.deleteProperty(key)});
  return {email:email,activated:true};
}
